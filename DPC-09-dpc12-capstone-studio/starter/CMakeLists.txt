cmake_minimum_required(VERSION 3.20)
project(AssetImport LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(importer importer.cpp)
add_executable(asset_import main.cpp)
target_link_libraries(asset_import PRIVATE importer)
include(CTest)
if(BUILD_TESTING)
    add_executable(import_checks checks.cpp)
    target_link_libraries(import_checks PRIVATE importer)
    foreach(test IN ITEMS standard malformed limit isolation comma mixed)
        add_test(NAME ${test} COMMAND import_checks ${test})
    endforeach()
    set_tests_properties(standard malformed limit isolation PROPERTIES LABELS baseline)
    set_tests_properties(comma mixed PROPERTIES LABELS feature)
endif()
